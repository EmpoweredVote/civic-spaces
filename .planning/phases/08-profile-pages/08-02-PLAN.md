     const [isProfileRoute] = useRoute('/profile/:userId')
     return (
       <>
         <Route path="/profile/:userId">

         <div style={{ display: isProfileRoute ? 'none' : undefined }} className="flex flex-col h-screen">

         </div>
       </>
     )
   }
