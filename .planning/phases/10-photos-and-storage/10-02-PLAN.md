<background>
Treasury Tracker (`C:/treasury-tracker/src/utils/wikiImage.ts`) already fetches municipality hero images from the Wikipedia REST API:
  GET https://en.wikipedia.org/api/rest_v1/page/summary/{article_title}
  → response.originalimage.source (preferred) or response.thumbnail.source

The `getHeroImage()` function takes a Municipality with `name`, `state`, and `entity_type`.
Our SliceInfo only has `sliceType` and `geoid` (TIGER/Line GEOIDs). We need a decoder.

Geoid formats used in our seed data:
  '1807'   — IN-07 congressional district (state_fips + district_num, 4 chars)
  '18046'  — IN State Senate District 46 (state_fips + zero-padded district, 5 chars)
  '18097'  — Monroe County, IN (standard 5-digit county FIPS)

Wikipedia title strategy per slice type:
  federal      → state name from first 2 digits (e.g. '1807' → '18' → 'Indiana')
  state        → state name from first 2 digits (e.g. '18046' → '18' → 'Indiana')
  local        → '{county} County, {state}' from 5-digit county FIPS (e.g. '18097' → 'Monroe County, Indiana')
  neighborhood → null (census tract GEOIDs have no Wikipedia article, use sliceCopy default)
  unified      → null (no geo identity, use sliceCopy default)
  volunteer    → null (no geo identity, use sliceCopy default)

Wikipedia returns beautiful lead photos for US states and counties — state capitols, skylines, landmarks.
Monroe County, Indiana's Wikipedia article features the Monroe County Courthouse.
Indiana's Wikipedia article features the Indiana State House.

/**
 * Derive a Wikipedia article title from a slice's type and TIGER/Line geoid.
 *
 * Returns null for slice types with no meaningful geographic article
 * (neighborhood census tracts, unified, volunteer).
 *
 * Mirrors the title-building logic in Treasury Tracker's wikiImage.ts,
 * adapted for TIGER/Line GEOIDs instead of named Municipality objects.
 */
export function geoidToWikiTitle(sliceType: SliceType, geoid: string): string | null {
  const stateFips = geoid.slice(0, 2)
  const stateName = STATE_FIPS[stateFips]

  switch (sliceType) {
    case 'federal':
      // Congressional district GEOIDs (e.g. '1807' = IN-07).
      // Fall back to state name — Wikipedia state articles have great landmark photos.
      return stateName ?? null

    case 'state':
      // State legislative district GEOIDs (e.g. '18046' = IN Senate 46).
      // Use state name — state capitol photos come from the state article.
      return stateName ?? null

    case 'local': {
      // Standard 5-digit county FIPS (e.g. '18097' = Monroe County, IN).
      if (geoid.length === 5 && stateName) {
        const countyFips = geoid.slice(2)
        const countyName = COUNTY_NAMES[`${stateFips}-${countyFips}`]
        if (countyName) return `${countyName} County, ${stateName}`
      }
      // Fallback: state name if county isn't in our lookup yet
      return stateName ?? null
    }

    case 'neighborhood':
    case 'unified':
    case 'volunteer':
    default:
      // Census tract GEOIDs are too granular for Wikipedia.
      // Unified/volunteer have no geographic identity.
      return null
  }
}

**Step 1 — Create `src/hooks/useWikiHeroImage.ts`:**

Mirrors Treasury Tracker's `getHeroImage()` pattern but as a React hook with session cache.

```typescript
import { useState, useEffect } from 'react'
import type { SliceInfo } from '../types/database'
import { geoidToWikiTitle } from '../lib/geoidToWiki'

/** Session-scoped cache: avoids re-fetching on tab switches within same session */
const cache = new Map<string, string | null>()

async function fetchWikiImage(title: string): Promise<string | null> {
  try {
    const encoded = encodeURIComponent(title.replace(/ /g, '_'))
    const resp = await fetch(
      `https://en.wikipedia.org/api/rest_v1/page/summary/${encoded}`,
      { headers: { Accept: 'application/json' } }
    )
    if (!resp.ok) return null
    const data = await resp.json()
    // Prefer originalimage for higher resolution hero banners (same as Treasury Tracker)
    return data.originalimage?.source ?? data.thumbnail?.source ?? null
  } catch {
    return null
  }
}

/**
 * Returns a Wikipedia hero image URL for a slice based on its type and geoid.
 * Returns null while loading or if no Wikipedia image is found for this slice type.
 *
 * Respects the same DB-override-first pattern as Treasury Tracker:
 * if slice.photoUrl is set, the caller should use it instead of this result.
 *
 * Results cached per session — same slice type+geoid never re-fetches.
 */
export function useWikiHeroImage(slice: SliceInfo): string | null {
  const [url, setUrl] = useState<string | null>(() => {
    // Synchronously return cached value on mount if available
    const cacheKey = `${slice.sliceType}|${slice.geoid}`
    return cache.has(cacheKey) ? (cache.get(cacheKey) ?? null) : null
  })

  useEffect(() => {
    const title = geoidToWikiTitle(slice.sliceType, slice.geoid)
    if (!title) return // neighborhood/unified/volunteer — no Wikipedia title

    const cacheKey = `${slice.sliceType}|${slice.geoid}`
    if (cache.has(cacheKey)) {
      setUrl(cache.get(cacheKey) ?? null)
      return
    }

    fetchWikiImage(title).then(result => {
      cache.set(cacheKey, result)
      setUrl(result)
    })
  }, [slice.sliceType, slice.geoid])

  return url
}
```
